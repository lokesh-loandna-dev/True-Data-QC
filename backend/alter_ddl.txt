-- 1. Add 'status' column to 'users' table (Registration & Approval feature)

-- ============================================================================

IF EXISTS (SELECT * FROM sysobjects WHERE name='users' AND xtype='U')

BEGIN

    IF NOT EXISTS (SELECT * FROM sys.columns WHERE object_id = OBJECT_ID('users') AND name = 'status')

    BEGIN

        ALTER TABLE users ADD status VARCHAR(50) NOT NULL DEFAULT 'inactive';

        -- Mark existing users as active so current accounts are not locked

        EXEC('UPDATE users SET status = ''active'' WHERE status = ''inactive''');

        PRINT 'Column status added to users table.';

    END

END

GO
 
-- ============================================================================

-- 2. Add 'review_required' column to 'loan_packages' table

-- ============================================================================

IF EXISTS (SELECT * FROM sysobjects WHERE name='loan_packages' AND xtype='U')

BEGIN

    IF NOT EXISTS (SELECT * FROM sys.columns WHERE object_id = OBJECT_ID('loan_packages') AND name = 'review_required')

    BEGIN

        ALTER TABLE loan_packages ADD review_required INT NULL DEFAULT 0;

        PRINT 'Column review_required added to loan_packages table.';

    END

END

GO
 
-- ============================================================================

-- 3. Create 'password_reset_otps' table (OTP Password Reset feature)

-- ============================================================================

IF OBJECT_ID('dbo.password_reset_otps', 'U') IS NULL

BEGIN

    CREATE TABLE dbo.password_reset_otps (

        id                  INT IDENTITY(1,1) PRIMARY KEY,

        email               VARCHAR(255) NOT NULL,

        otp_code            VARCHAR(10) NOT NULL,

        expires_at          DATETIME NOT NULL,

        verified            BIT NOT NULL DEFAULT 0,

        used                BIT NOT NULL DEFAULT 0,

        attempts            INT NOT NULL DEFAULT 0,

        created_at          DATETIME DEFAULT CURRENT_TIMESTAMP

    );

    CREATE INDEX IX_password_reset_otps_email ON dbo.password_reset_otps(email);

    PRINT 'Table password_reset_otps created successfully.';

END

GO
